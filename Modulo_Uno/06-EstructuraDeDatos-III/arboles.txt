function BinarySearchTree(value) {
  this.value = value,
  this.left = null,
  this.right = null
};

BinarySearchTree.prototype.size = function(){ 
  // Nodo Hoja
  if(!this.left && !this.rigth ) return 1;

   if(!this.left) return 1 + this.right.size();         // con un solo nodo ya sea el izq o el derecho pero solo uno
   if(!this.right) return 1 + this.left.size();
                                                       // con los dos nodos
    return 1 + this.left.size() + this.right.size();
};

BinarySearchTree.prototype.insert = function(value){  
  if (value > this.value){  // Deberia ir a la derecha
      if(!this.rigth){      // la posicion derecha esta libre
          this.right = new BinarySearchTree(value);
      }else {               // la posicion derecha esta ocupada
         this.right.insert(value);    // O sea como value > this.value va a la derecha si o si pero como ya esta ocupada se crea una nueva
      }
  }else{                   // deberia ir a la izquierda
    if(!this.left){
      this.left = new BinarySearchTree(value);   // se crea un nuevo nodo en caso que este vacia la posicion izquierda
    } else {            // en caso que ya este ocupado el nodo izquierdo se crea uno nuevo abajo 
      this.left.insert(value);
    }
  }
};


BinarySearchTree.prototype.contains = function(value){  

  if(this.value === value) return true;

  if(value > this.value) {
    if(!this.right){
      return false;
    } else {
      return this.rigth.contains(value);
    }
  } else {
    if(!this.left){
      return false;
    } else {
      return this.left.contains(value);
    }
  }
